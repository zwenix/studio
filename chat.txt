### Updated `src/ai/flows/grade1-math.ts`

```ts
// src/ai/flows/grade1-math.ts
import { defineFlow } from '@genkit-ai/flow';
import { googleAI, geminiProLatest, geminiFlashLatest } from '@genkit-ai/google-genai';
import { z } from 'zod';
import { db } from '@/lib/firebase';
import { collection, addDoc, serverTimestamp } from 'firebase/firestore';

export const generateGrade1Math = defineFlow(
  {
    name: 'generateGrade1Math',
    inputSchema: z.object({
      materialType: z.enum([
        'numberSense',
        'additionSubtraction',
        'shapesPatterns',
        'measurement',
        'money',
        'mathDisplay',
      ]),
      userId: z.string(),
    }),
    outputSchema: z.object({
      title: z.string(),
      content: z.string(),
      description: z.string(),
      assessmentCriteria: z.string(),   // New: CAPS-aligned assessment
      successIndicators: z.array(z.string()),
    }),
  },
  async ({ materialType, userId }) => {
    const capsPrompt = `You are an expert South African Foundation Phase Mathematics teacher.

Generate **CAPS-aligned Grade 1 Mathematics materials** with clear assessment criteria.

**Material Type:** ${materialType}

**CAPS Assessment Requirements (Must Include):**
- Informal assessment through observation
- Use of concrete apparatus and manipulatives
- Differentiation for different ability levels
- Clear success criteria (what "meeting expectations" looks like)
- Simple rubric or checklist for teachers
- Links to CAPS Learning Outcomes for Numbers, Patterns, Space & Shape, Measurement, Data Handling

**Output Requirements:**
- Child-friendly, engaging activities
- South African real-life contexts
- Teacher's Pet style (colourful, fun, simple)
- Include space for drawing, cutting, pasting where appropriate

Return clean JSON with:
{
  "title": "...",
  "content": "full markdown content with activities",
  "description": "short overview",
  "assessmentCriteria": "detailed CAPS assessment guidance for teachers",
  "successIndicators": ["observable success criteria 1", "success criteria 2", ...]
}`;

    let result;
    try {
      result = await geminiProLatest.generate({
        prompt: capsPrompt,
        config: { temperature: 0.65, maxOutputTokens: 5000 },
      });
    } catch (err) {
      console.warn('Gemini Pro failed, falling back to Flash...');
      result = await geminiFlashLatest.generate({
        prompt: capsPrompt,
        config: { temperature: 0.7, maxOutputTokens: 4200 },
      });
    }

    const output = JSON.parse(result.text);

    // Save to Firebase with CAPS metadata
    const docRef = await addDoc(collection(db, 'generatedContent'), {
      userId,
      subject: 'mathematics',
      type: materialType,
      title: output.title,
      content: output.content,
      description: output.description,
      assessmentCriteria: output.assessmentCriteria,
      successIndicators: output.successIndicators,
      createdAt: serverTimestamp(),
      modelUsed: 'gemini-pro-latest',
      capsAligned: true,
    });

    return {
      ...output,
      docId: docRef.id,
    };
  }
);
```
