  console.log(result.response.text());

import { getVertexAI, getGenerativeModel } from "firebase/vertexai";

const vertexAI = getVertexAI(app);
const model = getGenerativeModel(vertexAI, { model: "gemini-3.1-pro" });

async function generateIndividualizedPlan(studentDataObj) {
  // 1. Convert the JSON object to a string
  const studentDataString = JSON.stringify(studentDataObj, null, 2);

  // 2. Construct the prompt
  const prompt = `
    Analyze the following student data and generate a comprehensive 4-week 
    Individualized Development Plan (IDP). 
    
    Focus heavily on improving their specific weaknesses while utilizing their 
    stated learning styles and motivators. Output the plan in Markdown format 
    with clear weekly milestones.

    Student Data:
    ${studentDataString}
  `;

  // 3. Call the model
  try {
    const result = await model.generateContent(prompt);
    const idpMarkdown = result.response.text();
    console.log(idpMarkdown);
    return idpMarkdown;
    
  } catch (error) {
    console.error("Error generating IDP:", error);
  }
}

export async function generateEducationalContent(promptText) {

// Function for generating visual classroom posters
export async function generateClassroomPoster(topic) {
  try {
    const result = await imageModel.generateContent([
      `Create a high-resolution, vibrant, and educational classroom poster about: ${topic}. Ensure all text and labels are sharp, readable, and suitable for primary school learners.`
    ]);
    
    // The model returns the image as base64 data
    const base64Image = result.response.candidates[0].content.parts[0].inlineData.data;
    
    // Format it perfectly for a React Native <Image /> component
    return `data:image/png;base64,${base64Image}`;
  } catch (error) {
    console.error("Error generating poster:", error);
    throw error;
  }
}

import React, { useState } from 'react';
import { 
  View, 
  Text, 
  TextInput, 
  TouchableOpacity, 
  Image, 
  ScrollView, 
  ActivityIndicator, 
  StyleSheet 
} from 'react-native';

// Import the AI functions we created in the ai-services.js file
import { generateEducationalContent, generateClassroomPoster } from './ai-services'; 

export default function LessonGenerator() {
  // State management for user input and API responses
  const [topic, setTopic] = useState('');
  const [lessonPlan, setLessonPlan] = useState('');
  const [posterUri, setPosterUri] = useState(null);
  
  // State management for UI feedback
  const [isGenerating, setIsGenerating] = useState(false);
  const [error, setError] = useState('');

  const handleGenerate = async () => {
    if (!topic.trim()) return;

    // Reset UI states before starting a new generation
    setIsGenerating(true);
    setError('');
    setLessonPlan('');
    setPosterUri(null);

    try {
      // PRO TIP: Run both Gemini API calls at the same time using Promise.all
      // This cuts the waiting time for the user in half!
      const [textResult, imageResult] = await Promise.all([
        generateEducationalContent(`Create a structured lesson plan and IDP strategy for primary school learners about: ${topic}.`),
        generateClassroomPoster(topic)
      ]);

      // Update state with the results
      setLessonPlan(textResult);
      setPosterUri(imageResult);
      
    } catch (err) {
      console.error("Generation Failed:", err);
      setError('Failed to generate educational content. Please try again.');
    } finally {
      // Turn off the loading spinner
      setIsGenerating(false);
    }
  };

  return (
    <ScrollView style={styles.container} contentContainerStyle={styles.contentContainer}>
      <Text style={styles.headerTitle}>AI Lesson & Poster Studio</Text>
      
      {/* Input Section */}
      <View style={styles.inputContainer}>
        <TextInput
          style={styles.input}
          placeholder="Enter a topic (e.g., The Water Cycle, Fractions)..."
          value={topic}
          onChangeText={setTopic}
          editable={!isGenerating}
        />
        <TouchableOpacity 
          style={[styles.button, isGenerating && styles.buttonDisabled]} 
          onPress={handleGenerate}
          disabled={isGenerating}
        >
          <Text style={styles.buttonText}>
            {isGenerating ? 'Generating...' : 'Create Lesson & Poster'}
          </Text>
        </TouchableOpacity>
      </View>

      {/* Error Handling */}
      {error !== '' && (
        <Text style={styles.errorText}>{error}</Text>
      )}

      {/* Loading State */}
      {isGenerating && (
        <View style={styles.loadingContainer}>
          <ActivityIndicator size="large" color="#0066cc" />
          <Text style={styles.loadingText}>Crafting personalized learning materials...</Text>
        </View>
      )}

      {/* Results Section */}
      {!isGenerating && posterUri && (
        <View style={styles.resultCard}>
          <Text style={styles.sectionTitle}>Classroom Poster</Text>
          <Image 
            source={{ uri: posterUri }} 
            style={styles.posterImage} 
            resizeMode="contain"
          />
        </View>
      )}

      {!isGenerating && lessonPlan !== '' && (
        <View style={styles.resultCard}>
          <Text style={styles.sectionTitle}>Lesson Plan & Strategy</Text>
          {/* Note: The API returns Markdown. You may want a Markdown renderer here later */}
          <Text style={styles.lessonText}>{lessonPlan}</Text>
        </View>
      )}
      
    </ScrollView>
  );
}

// Replace the Pexels/Pixabay fetcher with Gemini Image Generation
async function fetchImage(query: string): Promise<string> {
  try {
    // Note: Ensure your Google GenAI plugin is configured to support Imagen
    const imageResponse = await ai.generate({
      model: googleAI.model('imagen-3.0-generate-001'), // The Google image generation model
      prompt: `Create a high-resolution, vibrant, and educational classroom poster or illustration about: ${query}. Ensure it is safe for primary school children.`,
      output: { format: 'media' }
    });
    
    // Returns the base64 URI for the React Native / Next.js frontend to render
    return imageResponse.media.url; 
  } catch (e) {
    console.error('AI Image generation failed for query:', query, e);
    return '';
  }
}
