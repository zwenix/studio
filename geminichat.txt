3. Implementation Example (Genkit Node.js)
Ensure you are using the latest plugin: @genkit-ai/google-genai (which replaced the older googleai package).

TypeScript
import { genkit } from 'genkit';
import { googleAI, gemini31Pro, gemini3Flash } from '@genkit-ai/google-genai';

const ai = genkit({
  plugins: [googleAI({ apiKey: process.env.GOOGLE_GENAI_API_KEY })],
  model: gemini3Flash, // Exported alias for 'gemini-3-flash-preview'
});

// Or using the string alias directly for flexibility:
const response = await ai.generate({
  model: 'googleai/gemini-pro-latest', 
  prompt: 'Generate a CAPS-compliant lesson plan for Grade 7 Maths.',
  config: {
    // New for Gemini 3.1: Control the "thinking" depth
    version: '3.1',
    thinkingLevel: 'medium', 
  },
});
