// src/ai/flows/grade1-english.ts
import { defineFlow, run } from '@genkit-ai/flow';
import { generate } from '@genkit-ai/ai';
import { gemini15Flash, groqLlama3 } from '@genkit-ai/ai/model'; // or your preferred model
import { z } from 'zod';

export const generateGrade1EnglishMaterials = defineFlow(
  {
    name: 'generateGrade1EnglishMaterials',
    inputSchema: z.object({
      materialType: z.enum(['phonicsBook', 'readingComprehensionA', 'spellingTr', 'displayPack', 'improvementTracker', 'handwritingSheet', 'abcBooklet', 'classroomLabels']),
      customPrompt: z.string().optional(),
    }),
    outputSchema: z.object({
      title: z.string(),
      content: z.string(),
      illustrations: z.array(z.string()), // URLs or base64
      style: z.literal('TeacherPetFont'),
    }),
  },
  async (input) => {
    const basePrompt = `You are an expert South African Foundation Phase teacher. Generate CAPS-aligned Grade 1 English materials in a colourful, hand-drawn style using Teacher's Pet Font. Use lots of graphic illustrations, bright colours, simple language.

Material type: ${input.materialType}
${input.customPrompt || ''}

Return JSON with:
{
  "title": "...",
  "content": "full markdown content with headings, tables, activities",
  "illustrations": ["https://... image url 1", "https://... image url 2"],
  "style": "TeacherPetFont"
}`;

    const result = await generate({
      model: groqLlama3, // or gemini15Flash if you prefer
      prompt: basePrompt,
      config: { temperature: 0.7 },
    });

    return JSON.parse(result.text);
  }
);

'use client';

import React, { useState } from 'react';
import { generateGrade1EnglishMaterials } from '@/ai/flows/grade1-english';
import { searchImages } from '@/tools/webTools'; // reuse your existing tool

export default function Grade1EnglishGenerator() {
  const [loading, setLoading] = useState(false);
  const [materials, setMaterials] = useState<any>(null);
  const [error, setError] = useState('');

  const generateMaterial = async (type: string) => {
    setLoading(true);
    setError('');

      setLoading(false);

      {materials && (

// src/components/Grade1EnglishGenerator.tsx
'use client';

import React, { useState } from 'react';
import { generateGrade1English } from '@/ai/flows/grade1-english';
import { db } from '@/firebase';
import { collection, addDoc, serverTimestamp } from 'firebase/firestore';
import { useAuth } from '@/context/AuthContext'; // Adjust path if needed

const materialOptions = [
  { id: 'phonicsBook', label: 'Phonics Book 1 (Full Booklet)' },
  { id: 'readingComprehensionA', label: "Reading Comprehension - 'a' Sound" },
  { id: 'spellingTr', label: "Spelling Practice - 'tr' Blends" },
  { id: 'displayPack', label: 'Classroom Display Pack (Alphabet Posters)' },
  { id: 'improvementTracker', label: 'Subject Improvement Plan Tracker' },
  { id: 'handwritingSheet', label: 'Handwriting Practice Sheet (Head, Tummy, Tail)' },
  { id: 'abcBooklet', label: 'ABC Handwriting Booklet' },
  { id: 'classroomLabels', label: 'Classroom Item Labels' },
];

export default function Grade1EnglishGenerator() {
  const { user } = useAuth();
  const [selectedType, setSelectedType] = useState('');
  const [loading, setLoading] = useState(false);
  const [result, setResult] = useState<any>(null);
  const [error, setError] = useState('');
  const [saved, setSaved] = useState(false);

  const generate = async () => {
    if (!selectedType || !user) return;
    setLoading(true);
    setError('');
    setResult(null);
    setSaved(false);

    try {
      const data = await generateGrade1English({ materialType: selectedType as any });

      setResult(data);

      // Save to Firebase
      await addDoc(collection(db, 'generatedContent', user.uid, 'grade1-english'), {
        type: selectedType,
        title: data.title,
        content: data.content,
        generatedAt: serverTimestamp(),
        materialType: selectedType,
      });

      setSaved(true);
    } catch (err: any) {
      setError(err.message || 'Generation failed');
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="max-w-6xl mx-auto">
      <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-4 gap-4 mb-10">
        {materialOptions.map((item) => (
          <button
            key={item.id}
            onClick={() => setSelectedType(item.id)}
            className={`p-6 text-left rounded-3xl transition-all ${
              selectedType === item.id
                ? 'bg-indigo-600 text-white shadow-xl'
                : 'bg-white dark:bg-gray-800 hover:bg-gray-100 dark:hover:bg-gray-700 border'
            }`}
          >
            <p className="font-semibold">{item.label}</p>
          </button>
        ))}
      </div>

      <button
        onClick={generate}
        disabled={!selectedType || loading || !user}
        className="w-full py-5 text-lg font-semibold rounded-3xl bg-gradient-to-r from-indigo-600 to-purple-600 text-white disabled:opacity-50"
      >
        {loading ? 'Generating with Gemini + Groq...' : 'Generate & Save to Library'}
      </button>

      {error && <p className="text-red-500 mt-6 text-center">{error}</p>}
      {saved && <p className="text-green-600 mt-6 text-center">✅ Saved to your content library!</p>}

      {result && (
        <div className="mt-12 bg-white dark:bg-gray-900 rounded-3xl shadow-2xl p-10">
          <h2 className="text-4xl font-bold mb-6" style={{ fontFamily: "'Comic Sans MS', cursive" }}>

            className="prose dark:prose-invert text-lg leading-relaxed"
